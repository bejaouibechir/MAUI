# 💥 **Exemple complet Shell avec naviguation (sans shell)**

---

## 🟢 **1️⃣ HomePage.xaml**

```xml
<?xml version="1.0" encoding="utf-8" ?>
<ContentPage xmlns="http://schemas.microsoft.com/dotnet/2021/maui"
             xmlns:x="http://schemas.microsoft.com/winfx/2009/xaml"
             x:Class="MauiApp1.HomePage"
             Title="Accueil">

    <StackLayout Padding="20">
        <Label Text="Bienvenue" FontSize="22" HorizontalOptions="Center"/>
        <Button Text="Aller à Détail" Clicked="OnDetailClicked"/>
    </StackLayout>
</ContentPage>
```

---

### ✅ HomePage.xaml.cs

```csharp
namespace MauiApp1
{
    public partial class HomePage : ContentPage
    {
        public HomePage()
        {
            InitializeComponent();
        }

        private async void OnDetailClicked(object sender, EventArgs e)
        {
            await Shell.Current.GoToAsync(nameof(DetailPage));
        }
    }
}
```

## 🟢 **3️⃣ AppShell.xaml**

```xml

<Shell
    xmlns="http://schemas.microsoft.com/dotnet/2021/maui"
    xmlns:x="http://schemas.microsoft.com/winfx/2009/xaml"
    xmlns:local="clr-namespace:MauiApp1"

    <Tab Title="Accueil" Icon="home.png">
        <ShellContent Title="Accueil" ContentTemplate="{DataTemplate local:HomePage}" />
    </Tab>
```

---

### ✅ AppShell.xaml.cs

```csharp
namespace MauiApp1
{
    public partial class AppShell : Shell
    {
        public AppShell()
        {
            InitializeComponent();

            // Enregistrer la route pour la page de détail
            Routing.RegisterRoute(nameof(DetailPage), typeof(DetailPage));
        }
    }
}
```

# 🟢 **2️⃣ DetailPage.xaml**

```xml
<?xml version="1.0" encoding="utf-8" ?>
<ContentPage xmlns="http://schemas.microsoft.com/dotnet/2021/maui"
             xmlns:x="http://schemas.microsoft.com/winfx/2009/xaml"
             x:Class="MauiApp1.DetailPage"
             Title="Détail">

    <StackLayout Padding="20">
        <Label Text="Page Détail" FontSize="22" HorizontalOptions="Center"/>
        <Button Text="Sous-page 1" Clicked="OnFirstSub"/>
        <Button Text="Sous-page 2" Clicked="OnSecondSub"/>
        <Button Text="Ouvrir Modale" Clicked="OnOpenModal"/>
    </StackLayout>
</ContentPage>
```

---

### ✅ DetailPage.xaml.cs

```csharp
namespace MauiApp1
{
    public partial class DetailPage : ContentPage
    {
        public DetailPage()
        {
            InitializeComponent();
        }

        private async void OnFirstSub(object sender, EventArgs e)
        {
            await Shell.Current.GoToAsync(nameof(FirstSubDetailPage));
        }

        private async void OnSecondSub(object sender, EventArgs e)
        {
            await Shell.Current.GoToAsync(nameof(SecondSubDetailPage));
        }

        private async void OnOpenModal(object sender, EventArgs e)
        {
            await Shell.Current.GoToAsync(nameof(ModalPage), true); // true = modal
        }
    }
}
```
