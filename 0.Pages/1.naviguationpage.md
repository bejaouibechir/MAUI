# 💥 Naviguation Page Labo

``` mermaid

graph TD
    A[MainPage] -->|PushAsync| B[DetailPage]
    B -->|PushAsync| C[ExtraDetailPage]
    A -->|PushModalAsync| D[ModalPage]
    A -->|InsertPageBefore| E[InsertedPage]
    A -->|RemovePage| F[Suppression deuxième page]

    B -->|PopAsync| A
    C -->|PopAsync| B
    D -->|PopModalAsync| A

```

## ✅ MainPage.xaml

```xml
<?xml version="1.0" encoding="utf-8" ?>
<ContentPage xmlns="http://schemas.microsoft.com/dotnet/2021/maui"
             x:Class="MauiApp1.MainPage"
             Title="Accueil"
             BackgroundColor="Beige">

    <StackLayout Padding="20" Spacing="15">
        <Button x:Name="BtnToDetail" Text="Aller à Détails (PushAsync)" />
        <Button x:Name="BtnToModal" Text="Ouvrir Modal (PushModalAsync)" />
        <Button x:Name="BtnInsertBefore" Text="Insérer une page avant racine" />
        <Button x:Name="BtnRemovePage" Text="Supprimer deuxième page" />
    </StackLayout>
</ContentPage>
```

---

## ✅ MainPage.xaml.cs

```csharp
namespace MauiApp1
{
    public partial class MainPage : ContentPage
    {
        public MainPage()
        {
            InitializeComponent();

            BtnToDetail.Clicked += async (s, e) =>
            {
                await Navigation.PushAsync(new DetailPage());
            };

            BtnToModal.Clicked += async (s, e) =>
            {
                await Navigation.PushModalAsync(new ModalPage());
            };

            BtnInsertBefore.Clicked += (s, e) =>
            {
                var insertedPage = new InsertedPage();
                Navigation.InsertPageBefore(insertedPage, Navigation.NavigationStack[0]);
                DisplayAlert("Info", "Page insérée avant la racine.", "OK");
            };

            BtnRemovePage.Clicked += (s, e) =>
            {
                if (Navigation.NavigationStack.Count > 1)
                {
                    var pageToRemove = Navigation.NavigationStack[1];
                    Navigation.RemovePage(pageToRemove);
                    DisplayAlert("Info", "Deuxième page supprimée.", "OK");
                }
                else
                {
                    DisplayAlert("Info", "Pas de deuxième page à supprimer.", "OK");
                }
            };
        }
    }
}
```
