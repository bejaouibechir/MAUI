# ✅ **3️⃣ AppShell.xaml.cs**

```csharp
using System.Diagnostics;
using System.IO;
using System.Text;

namespace MauiApp1;

public partial class AppShell : Shell
{
    private MemoryStream _memoryStream = new();

    public AppShell()
    {
        InitializeComponent();
    }

    public void SaveData(string login, string password, string role)
    {
        _memoryStream.SetLength(0);
        using var writer = new StreamWriter(_memoryStream, Encoding.UTF8, leaveOpen: true);
        writer.WriteLine(login);
        writer.WriteLine(password);
        writer.WriteLine(role);
        writer.Flush();
        Debug.WriteLine($"[SaveData] Login: {login}, Password: {password}, Role: {role}");
    }

    public (string login, string password, string role) LoadData()
    {
        _memoryStream.Seek(0, SeekOrigin.Begin);
        using var reader = new StreamReader(_memoryStream, Encoding.UTF8, leaveOpen: true);
        var login = reader.ReadLine() ?? string.Empty;
        var password = reader.ReadLine() ?? string.Empty;
        var role = reader.ReadLine() ?? string.Empty;
        Debug.WriteLine($"[LoadData] Login: {login}, Password: {password}, Role: {role}");
        return (login, password, role);
    }

    protected override void OnNavigating(ShellNavigatingEventArgs args)
    {
        base.OnNavigating(args);
        Debug.WriteLine($"[OnNavigating] From: {args.Source}, To: {args.Target.Location}");

        if (args.Target.Location.OriginalString == "//detailsPage")
        {
            var (_, _, role) = LoadData();

            if (!string.Equals(role, "admin", StringComparison.OrdinalIgnoreCase))
            {
                Debug.WriteLine("[OnNavigating] Accès refusé : rôle non autorisé.");
                args.Cancel();

                MainThread.BeginInvokeOnMainThread(async () =>
                {
                    await Application.Current.MainPage.DisplayAlert("Accès refusé", "Vous devez être administrateur pour accéder à cette page.", "OK");
                });
            }
        }
    }

    protected override void OnNavigated(ShellNavigatedEventArgs args)
    {
        base.OnNavigated(args);
        Debug.WriteLine($"[OnNavigated] From: {args.Previous?.Location}, To: {args.Current.Location}");
    }
}
```

# ✅ **5️⃣ HomePage.xaml.cs**

```csharp
using Microsoft.Maui.Controls;

namespace MauiApp1.Pages
{
    public partial class HomePage : ContentPage
    {
        public HomePage()
        {
            InitializeComponent();
        }

        private async void OnDetailsClicked(object sender, EventArgs e)
        {
            if (Shell.Current is AppShell appShell)
            {
                string login = LoginEntry.Text;
                string password = PasswordEntry.Text;
                string role = RoleEntry.Text;

                appShell.SaveData(login, password, role);
            }

            await Shell.Current.GoToAsync("//detailsPage");
        }
    }
}
```

# ✅ **7️⃣ DetailsPage.xaml.cs**

```csharp
using Microsoft.Maui.Controls;

namespace MauiApp1.Pages
{
    public partial class DetailsPage : ContentPage
    {
        public DetailsPage()
        {
            InitializeComponent();
        }

        protected override void OnAppearing()
        {
            base.OnAppearing();

            if (Shell.Current is AppShell appShell)
            {
                var (login, password, _) = appShell.LoadData();
                LoginLabel.Text = login;
                PasswordLabel.Text = password;
            }
        }

        private async void OnBackClicked(object sender, EventArgs e)
        {
            await Shell.Current.GoToAsync("//homePage");
        }
    }
}
```

# ✅ **🎯 Résultat final**

* L’utilisateur saisit **login**, **password**, et **role** (admin ou user) sur la HomePage.
* Les données sont **sauvegardées** dans la mémoire.
* Lors de la tentative d’accès à DetailsPage :

  * ✅ Si le rôle = **admin** → accès autorisé et infos affichées.
  * ❌ Si le rôle ≠ admin → **accès refusé**, alerte affichée, navigation annulée.
